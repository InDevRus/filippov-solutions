    $\br{\pow{x}{2} + \pow{y}{2}} \parder{z} {x} + 2xy \parder{z} {y} = -\pow{z}{2}$.
    
    $\dvfrac {\di x} {\pow{x}{2} + \pow{y}{2}}
    = \dvfrac {\di y} {2xy} = \dvfrac {\di z} {-\pow{z}{2}}$.
    
    \begin{enumerate}
        \item $\dvfrac {\di x} {\pow{x}{2} + \pow{y}{2}} = \dvfrac {\di y} {2xy}$.
        $\der{x}{y} = \dvfrac {\pow{x}{2} + \pow{y}{2}} {2xy}$.
        $2x\'\br{y} = \dvfrac{x}{y} + \dvfrac {y} {x}$.
        Получилось однородное уравнение, заменяем
        $x\br{y} = y \cdot w\br{y}$.
        $2w\'y + 2w = w + \dvfrac {1} {w}$.
        $2w\'y = \dvfrac {1 - \pow{w}{2}} {w}$.
        $\dvfrac {2w} {1 - \pow{w}{2}} w\' = \dvfrac {1} {y}$.
        $-\ln\vbr{1 - \pow{w}{2}} = \ln\vbr{y} + A$.
        $\br{1 - \pow{w}{2}} y = A$.
        $\dvfrac {\pow{y}{2} - \pow{x}{2}} {y} = A$.
        
        \item $\dvfrac {\di x + \di y} {\pow{x}{2} + \pow{y}{2} + 2xy}
        = \dvfrac {\di z} {-\pow{z}{2}}$.
        $\dvfrac {d\br{x + y}} {\br{x + y}^2} 
        = -d\br{\dvfrac {1} {x + y}} = d\br{\dvfrac {1} {z}}$.
        $\dvfrac {1} {z} + \dvfrac {1} {x + y} = B$.
    \end{enumerate}
    
    Окончательно,
    $F\br{\dvfrac {\pow{y}{2} - \pow{x}{2}} {y}; \dvfrac {1} {z} + \dvfrac {1} {x + y}} = 0$.
    Выразим решение также в явном виде:
    $\dvfrac {1} {z} + \dvfrac {1} {x + y} = f\br{\dvfrac {\pow{y}{2} - \pow{x}{2}} {y}}$.
    $z = \br{f\br{\dvfrac {\pow{y}{2} - \pow{x}{2}} {y}} - \dvfrac {1} {x + y}}^{-1}$.
