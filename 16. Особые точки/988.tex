\begin{framed}
    \begin{proposition}\end{proposition}
    При $r\br{\xi; \phi} = \sqrt{\pow{\xi}{2} + \pow{\phi}{2}} \to 0$
    имеем $\xi \phi = O\br{\pow{r}{2}}$.
    В частности
    $$\dfrac{\xi \phi} {\pow{r}{1 + \epsilon}} 
    = \dfrac {\gamma\br{r} \pow{r}{2}} {\pow{r}{1 + \epsilon}} \to 0
    \text{ для всех }\epsilon \in \br{0; 1}.$$
    
    Действительно, 
    $0 \le \br{\vbr{\xi} - \vbr{\phi}}^2$
    и 
    $\vbr{\xi \phi} 
    \le \dfrac {1} {2} \br{\pow{\xi}{2} + \pow{\phi}{2}}
    \le \dfrac {1} {2} \br{\pow{r}{2} + \pow{r}{2}}
    = \pow{r}{2}$,
    что показывает $\xi \phi = O\br{\pow{r}{2}}$.
\end{framed}

$\tsys{
    P\br{x; y} = \sqrt{\pow{x}{2} - y + 2} - 2
    \\ Q\br{x; y} = \arctg\br{\pow{x}{2} + xy}
}$;
$\tsys{
    \sqrt{\pow{x}{2} - y + 2} - 2 = 0
    \\ \arctg\br{\pow{x}{2} + xy} = 0
}$;
$\tsys{
    y = \pow{x}{2} - 2
    \\ \pow{x}{2} + xy = 0
}$;
$x \br{x + \pow{x}{2} - 2} = 0$.
$\tsys{
    \sub{x}{1} = 0
    \\ \sub{y}{1} = -2
}$,
$\tsys{
    \sub{x}{2} = 1
    \\ \sub{y}{2} = -1
}$,
$\tsys{
    \sub{x}{3} = -2
    \\ \sub{y}{3} = 2
}$, и особые точки -- $\sub{A}{1}\br{0\; -2}$, $\sub{A}{2}\br{1; -1}$ и $\sub{A}{3} = \br{-2; 2}$.

Будем также пользоваться тем, что $\sqrt{1 + u} = 1 + \dfrac {1} {2}u + O\br{\pow{u}{2}}$ и $\arctg\br{u} = u + O\br{\pow{u}{3}}$ при $u \to 0$.

\begin{enumerate}
    \item $y = \phi - 2$. \\
        $\begin{aligned}
        \dot{x} = \sqrt{\pow{x}{2} + \phi + 4} - 2
        & = 2\sqrt{1 + \dfrac {\pow{x}{2}} {4} - \dfrac {\phi} {4}} - 2 \\
        & = 2\ \br{1 + \dfrac {1} {2} \br{\dfrac {\pow{x}{2}} {4} - \dfrac {\phi} {4}} 
            + O\br{\br{\dfrac {\pow{x}{2}} {4} - \dfrac {\phi} {4}}^2}} - 2 = \\
        & = - \dfrac {\phi} {4} + O\br{\pow{r}{2}}.
    \end{aligned}$
    
    $\dot{\phi} = \arctg\br{\pow{x}{2} + x\phi - 2x} 
    = \pow{x}{2} + x\phi - 2x + O\br{\br{\pow{x}{2} + x\phi - 2x}^3} 
    = -2x + O\br{\pow{r}{2}}$.
    $A = \begin{pmatrix} 0 & - \frac {1} {4} \vspace{1mm}  \\ -2 & \hspace{2mm} 0 \end{pmatrix}$.
    $\pow{\lambda}{2} - \dfrac {1} {2} = 0$.
    $\sub{\lambda}{1} = -\dfrac {1} {\sqrt{2}}$,
    $\sub{\lambda}{2} = \dfrac {1} {\sqrt{2}}$,
    а особая точка $\sub{A}{1}\br{0; -2}$ -- <<седло>>.
    
    Найдём сепаратрисы. 
    $A - \sub{\lambda}{1} E \sim \begin{pmatrix} -2\sqrt{2} & 1\end{pmatrix}$,
    $\sub{v}{1} = \begin{pmatrix} 1 \\ 2\sqrt{2} \end{pmatrix}$,
    $\sub{\phi}{1} = 2\sqrt{2} x$.
    Аналогично,
    $A - \sub{\lambda}{2} E \sim \begin{pmatrix} 2\sqrt{2} & 1\end{pmatrix}$,
    $\sub{v}{2} = \begin{pmatrix} 1 \\ -2\sqrt{2} \end{pmatrix}$,
    $\sub{\phi}{2} = -2\sqrt{2} x$.
    \pagebreak
    
    \item $\tsys{
        x = \xi + 1 \\
        y = \phi - 1
    }$.
    
    $\begin{aligned}
        \dot{\xi} 
        & = \sqrt{\pow{\xi}{2} + 2\xi - \phi + 4} - 2
        = 2\sqrt{1 + \dfrac {\pow{\xi}{2}} {4} + \dfrac {\xi} {2} - \dfrac {\phi} {4}} - 2 \\
        & = 2\ \br{1 + \dfrac {1}{2} 
            \br{\dfrac {\pow{\xi}{2}} {4} + \dfrac {\xi} {2} - \dfrac {\phi} {4} } 
            + O\br{\br{\dfrac {\pow{\xi}{2}} {4} + \dfrac {\xi} {2} - \dfrac {\phi} {4}}^2} }
            - 2 \\
        & = \dfrac {\xi} {2} - \dfrac {\phi} {4} + O\br{\pow{r}{2}}.\end{aligned}$
    
    $\dot{\phi} = \arctg\br{\br{\xi + 1}\br{\xi + \phi}} 
    = \pow{\xi}{2} + \xi + \xi\phi + \phi + O\br{\pow{r}{3}} 
    = \xi + \phi + O\br{\pow{r}{2}}$.
    
    $A = \begin{pmatrix}
        \frac {1} {2} \vspace{1mm} & - \frac {1} {4} \\
        1 & \hspace{2mm} 1 \end{pmatrix}$.
    $\sub{\lambda}{1, 2} = \dfrac {3 \mp \sqrt{3} i} {4}$,
    и особая точка $\sub{A}{2}\br{1; -1}$
    является <<фокусом>>, причём неустойчивым, то есть траектории раскручиваются.
    
    В $\tsys{\xi = -0,6 \\ \phi = 0,6}$ для 
    $\tsys{
        \dot{\xi} = \sqrt{\pow{\xi}{2} + 2\xi - \phi + 4} - 2
        \\ \dot{\phi} = \arctg\br{\br{\xi + 1}\br{\xi + \phi}}
    }$ имеем вектор 
    $\tsys{
        \dot{\xi}\br{-0,6; 0,6} = -0,4 \\
        \dot{\phi}\br{-0,6; 0,6} = 0}.$
    А следовательно, направление раскручивания траекторий -- против часовой.
    
    \item $\tsys{
        x = \xi - 2 \\
        y = \phi + 2
    }$.
    
    $\begin{aligned}
        \dot{\xi} & = \sqrt{\pow{\xi}{2} - 4\xi - \phi + 4} - 2 \\
        & = 2\ \br{1 + \dfrac {1} {2} \br{\dfrac {\pow{\xi}{2}} {4} 
        - \xi - \dfrac {\phi} {4}} + O\br{\pow{r}{2}}} - 2 \\
        & = -\xi - \dfrac {\phi} {4} + O\br{\pow{r}{2}}.
    \end{aligned}$
    
    $\dot{\phi} = \arctg\br{\br{\xi - 2} \br{\xi + \phi}}
    = \pow{\xi}{2} - 2\xi + \xi \phi - 2\phi + O\br{\pow{r}{3}}
    = -2\xi - 2\phi + O\br{\pow{r}{2}}$.
    $A = \begin{pmatrix} -1 \vspace{1mm} & - \frac {1} {4}  \\ 1 & \hspace{2mm} 1 \end{pmatrix}$.
    $\sub{\lambda}{1} = \dfrac {-3 - \sqrt{3}} {2}$.
    $\sub{\lambda}{2} = \dfrac {-3 + \sqrt{3}} {2}$.
    $\sub{\lambda}{1} < \sub{\lambda}{2} < 0$,
    поэтому особая точка $\sub{A}{3}\br{-2; 2}$
    -- <<узел>>, при этом устойчивый.
    В полной аналогии с предыдущими случаями находим
    $\sub{v}{1} = \begin{pmatrix} 1 \\ 2 + 2\sqrt{3} \end{pmatrix}$,
    $\sub{v}{2} = \begin{pmatrix} 1 \\ 2 - 2\sqrt{3} \end{pmatrix}$;
    $\sub{\phi}{1} = \br{2 + 2\sqrt{3}} \xi$ -- трансверсаль, а
    $\sub{\phi}{2} = \br{2 - 2\sqrt{3}} \xi$ -- касательная.
\end{enumerate}
