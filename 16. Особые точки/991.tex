\begin{framed}
    \begin{proposition}\end{proposition}
    При $r\br{\xi; \phi} = \sqrt{\pow{\xi}{2} + \pow{\phi}{2}} \to 0$
    имеем $\xi \phi = O\br{\pow{r}{2}}$.
    В частности
    $$\dfrac{\xi \phi} {\pow{r}{1 + \epsilon}}
    = \dfrac {\gamma\br{r} \pow{r}{2}} {\pow{r}{1 + \epsilon}} \to 0
    \text{ для всех } \epsilon \in \br{0; 1}.$$
    
    Действительно, 
    $0 \le \br{\vbr{\xi} - \vbr{\phi}}^2$
    и 
    $\vbr{\xi \phi}     
    \le \dfrac {1} {2} \br{\pow{\xi}{2} + \pow{\phi}{2}}
    \le \dfrac {1} {2} \br{\pow{r}{2} + \pow{r}{2}}
    = \pow{r}{2}$,
    что показывает $\xi \phi = O\br{\pow{r}{2}}$.
\end{framed}

$\tsys{
    & P\br{x; y} = \ln\br{1 - y + \pow{y}{2}}
    \\& Q\br{x; y} = 3 - \sqrt{\pow{x}{2} + 8y}
}$;
$\tsys{
    & \pow{y}{2} - y = 0
    \\& \pow{x}{2} = 9 - 8y
}$.\\
Итак, получаем 4 особые точки:
$\sub{A}{1}\br{-3; 0}$, $\sub{A}{2}\br{-1; 1}$, $\sub{A}{3}\br{1; 1}$, $\sub{A}{4}\br{3; 0}$.

Для оптимизации процесса разложим каждую функцию заранее. Далее будут применены замены вида
$\tsys{& x = \sub{\xi}{i} + \sub{x}{i} \\& y = \sub{\phi}{i} + \sub{y}{i}}$,
где $\sub{A}{i}\br{\sub{x}{i}; \sub{y}{i}}$, $1 \le i \le 4$.
Предположим, что $\sub{\xi}{i} \to 0$ и $\sub{\phi}{i} \to 0$.
Тогда $\pow{y}{2} - y \to 0$ и $\pow{x}{2} + 8y \to 9$, $O\br{\br{x - \sub{x}{i}}^k} 
= O\br{\subpow{\xi}{i}{k}} = O\br{\br{O\br{\sub{r}{i}}}^k} = O\br{\subpow{r}{i}{k}}$
для натуральных $k$, и аналогично для $y - \sub{y}{i}$ и $\sub{\phi}{i}$.

\begin{itemize}
    \item $\begin{aligned}[t]
        \ln\br{1 - y + \pow{y}{2}}
        & = -y + \pow{y}{2} + O\br{\br{-y + \pow{y}{2}}^2}
            = -y + \pow{y}{2} + O\br{\subpow{r}{i}{2}} = \\
        & = -\sub{\phi}{i} - \sub{y}{i} + \subpow{\phi}{i}{2} 
            + 2\sub{\phi}{i}\,\sub{y}{i} + \subpow{y}{i}{2} 
            + O\br{\subpow{r}{i}{2}} = \\
        & = \br{\subpow{y}{i}{2} - \sub{y}{i}} + \br{2\sub{y}{i} - 1}\sub{\phi}{i} 
            + O\br{\subpow{r}{i}{2}} = \\
        & = \br{2\sub{y}{i} - 1}\sub{\phi}{i} + O\br{\subpow{r}{i}{2}},
    \end{aligned},$
    
    так как $\subpow{y}{i}{2} - \sub{y}{i} = 0$;
    
    \item $\begin{aligned}[t]
        3 - \sqrt{\pow{x}{2} + 8y}
        & = 3 - 3\sqrt{1 + \br{\dfrac {1} {9} \br{\pow{x}{2} + 8y} - 1}} = \\
        & = - 3\br{1 + \dfrac {1}{2} \br{\dfrac {1}{9} \br{\pow{x}{2} + 8y} - 1} 
            + O\br{\subpow{r}{i}{2}}} \linebreak + 3 = \\
        & = -\dfrac {1}{6} \br{\pow{x}{2} + 8y} + \dfrac {3} {2} 
            + O\br{\subpow{r}{i}{2}} = \\
        & = -\dfrac {1}{6} \br{\subpow{x}{i}{2} + 2\sub{\xi}{i} \sub{x}{i} 
            + \subpow{\xi}{i}{2} + 8\sub{\phi}{i} + 8\sub{y}{i} } + \dfrac {3} {2}
            + O\br{\subpow{r}{i}{2}} = \\
        & = -\dfrac {1}{6} \br{\br{\subpow{x}{i}{2} 
            + 8\sub{y}{i}} + 2\sub{\xi}{i} \sub{x}{i} + 8\sub{\phi}{i} } + \dfrac {3} {2} 
            + O\br{\subpow{r}{i}{2}} = \\
        & = -\dfrac {1} {3} \sub{x}{i} \sub{\xi}{i} - \dfrac {4} {3} \sub{\phi}{i} 
            + O\br{\subpow{r}{i}{2}}, \text{ поскольку } \subpow{x}{i}{2} + 8\sub{y}{i} = 9.
        \end{aligned}$
\end{itemize}

Итак, после всякой замены 
$\tsys{& x = \sub{\xi}{i} + \sub{x}{i} \\& y = \sub{\phi}{i} + \sub{y}{i}}$
имеем $\tsys{
    & \dot{\xi}\br{t} = \br{2\sub{y}{i} - 1}\sub{\phi}{i} + O\br{\subpow{r}{i}{2}} 
    \\& \dot{\phi}\br{t} = -\dfrac {1} {3} \sub{x}{i} \sub{\xi}{i} - \dfrac {4} {3} \sub{\phi}{i} 
        + O\br{\subpow{r}{i}{2}}}
$.

\begin{enumerate}
    \item $\tsys{& \sub{x}{1} = -3 \\& \sub{y}{1} = 0}$; $\tsys{& x = \xi - 3 \\& y = \phi}$.
    $\tsys{
        & \dot{\xi} = -\phi + O\br{\pow{r}{2}} 
        \\& \dot{\phi} = \xi - \dfrac {4} {3} \phi + O\br{\pow{r}{2}}
    }$. \\
    $A = \begin{pmatrix} 0 &  -1 \\ 1 & -\frac {4} {3} \end{pmatrix}$.
    $\pow{\lambda}{2} - \dfrac {4} {3} \lambda + 1 = 0$, то есть
    $\sub{\lambda}{1, 2} = \dfrac {-2 \mp \sqrt{5}i} {3}$,
    а особая точка $\sub{A}{1}\br{-3; 0}$ -- <<фокус>>, причём устойчивый.
    
    Система до линеаризации выглядит, как
    $\tsys{
        & \dot{\xi} = \ln\br{1 + \pow{\phi}{2} - \phi} 
        \\& \dot{\phi} = 3 - \sqrt{\pow{\xi}{2} - 6\xi + 8\phi + 9}
    }$, а в частности
    $\dot{\xi}\br{\dfrac {1} {4}; 0} = 0$ и 
    $\dot{\phi}\br{\dfrac {1} {4}; 0} = \dfrac {1} {4}$,
    то есть направления закручивания -- против часовой.
    
    \item $\tsys{& \sub{x}{2} = -1 \\& \sub{y}{2} = 1}$;
    $\tsys{& x = \xi - 1 \\& y = \phi + 1}$.
    $\tsys{
        & \dot{\xi} = \phi + O\br{\pow{r}{2}} 
        \\& \dot{\phi} = \dfrac {1} {3} \xi - \dfrac {4} {3} \phi + O\br{\pow{r}{2}}
    }$. \\
    $A = \begin{pmatrix} 0 &  1 \\ \frac {1} {3} & -\frac {4} {3} \end{pmatrix}$.
    $\pow{\lambda}{2} + \dfrac {4} {3} \lambda - \dfrac {1} {3} = 0$.
    $\sub{\lambda}{1, 2} = \dfrac {-2 \mp \sqrt{7}} {3}$,
    особая точка $\sub{A}{2}\br{-1; 1}$ -- <<седло>>.
    $\sub{v}{1} = \begin{pmatrix} -3 \\ 2 + \sqrt{7}\end{pmatrix}$,
    $\sub{v}{2} = \begin{pmatrix} -3 \\ 2 - \sqrt{7}\end{pmatrix}$, 
    а $\phi = \dfrac {-2 -\sqrt{7}} {3}\, \xi$
    и $\phi = \dfrac {-2 +\sqrt{7}} {3}\, \xi$ -- сепаратрисы.
    
    \item $\tsys{& \sub{x}{3} = 1 \\& \sub{y}{3} = 1}$;
    $\tsys{& x = \xi + 1 \\& y = \phi + 1}$.
    $\tsys{
        & \dot{\xi} = \phi + O\br{\pow{r}{2}} 
        \\& \dot{\phi} = -\dfrac {1} {3} \xi - \dfrac {4} {3} \phi + O\br{\pow{r}{2}}
    }$. \\
    $A = \begin{pmatrix} 0 &  1 \\ -\frac {1} {3} & -\frac {4} {3} \end{pmatrix}$.
    $\pow{\lambda}{2} + \dfrac {4} {3} \lambda + \dfrac {1} {3} = 0$.
    $\sub{\lambda}{1} = -1$, $\sub{\lambda}{2} = -\dfrac {1} {3}$,
    особая точка $\sub{A}{3}\br{1; 1}$ -- <<узел>>, причём устойчивый.
    
    $\begin{aligned}
        & A - \sub{\lambda}{1} E
        = \begin{pmatrix} 1 & 1 \\ -\frac {1} {3} & -\frac {1} {3} \end{pmatrix}
        \sim \begin{pmatrix} 1 & 1 \end{pmatrix}.
        & \sub{v}{1} = \begin{pmatrix} 1 \\ -1 \end{pmatrix} \\        
        & A - \sub{\lambda}{2} E
        = \begin{pmatrix} \frac {1} {3} \vspace{1mm} & 1 \\ -\frac {1} {3} & 1 \end{pmatrix}
        \sim \begin{pmatrix} 1 & 3 \end{pmatrix}.
        & \sub{v}{2} = \begin{pmatrix} 3 \\ -1 \end{pmatrix}
    \end{aligned}$\\
    Следовательно, $\phi = -\xi$ -- трансверсаль,
    $\phi = -\dfrac {1} {3} \xi$ -- касательная.
    \pagebreak
    
    \item $\tsys{& \sub{x}{4} = 3 \\& \sub{y}{4} = 0}$;
    $\tsys{& x = \xi + 3 \\& y = \phi}$.
    $\tsys{
        & \dot{\xi} = -\phi + O\br{\pow{r}{2}} 
        \\& \dot{\phi} = -\xi - \dfrac {4} {3} \phi + O\br{\pow{r}{2}}
    }$. \\
    $A = \begin{pmatrix} 0 & -1 \\ -1 & -\frac {4} {3} \end{pmatrix}$.
    $\pow{\lambda}{2} + \dfrac {4} {3} \lambda - 1 = 0$.
    $\sub{\lambda}{1, 2} = \dfrac {-2 \mp \sqrt{13}} {3}$,
    особая точка $\sub{A}{4}\br{3; 0}$ -- <<седло>>.
    $\sub{v}{1} = \begin{pmatrix} 3 \\ 2 + \sqrt{13} \end{pmatrix}$,
    $\sub{v}{2} = \begin{pmatrix} 3 \\ 2 - \sqrt{13} \end{pmatrix}$. 
    $\phi = \dfrac {\sqrt{13} + 2} {3}\,\xi$ и
    $\phi = \dfrac {\sqrt{13} - 2} {3}\,\xi$ -- сепаратрисы.
\end{enumerate}
