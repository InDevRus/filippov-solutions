Для каждой линии будем искать дифференциальное уравнение
путём выражения произвольной постоянной $C$ и последующего 
дифференцирования. Затем будем подставлять 
$y\`\br{x} = - \dfrac {1} {z\`\br{x}}$, где $z\br{x}$ 
-- искомые ортогональные траектории 
(см. задачи \textbf{37} -- \textbf{50}).
В первых двух также использован тот факт, что 
$\tint 2z\br{x}z\`\br{x} \di x = \tint 2z \di z = \linebreak
= \pow{z}{2}\br{x} + C$.

\begin{enumerate}
    \item $y = C\pow{x}{2}$.
    $C = \dfrac {y} {\pow{x}{2}}$.
    $0 = \dfrac {y\` \cdot \pow{x}{2} - y \cdot 2x} {\pow{x}{4}}$.
    $xy\` - 2y = 0$.
    $- \dfrac {x} {z\`} = 2z$.
    $z\` = -\dfrac {x} {2z}$.
    $2zz\` = -x$.
    $\pow{z}{2} = -\dfrac {1}{2}\pow{x}{2} + C$.
    $2\pow{z}{2} + \pow{x}{2} = C$.

    \item $y = C\pow{e}{x}$.
    $C = y \pow{e}{-x}$.
    $0 = y\` \pow{e}{-x} - y \pow{e}{-x}$.
    $y\` = y$.
    $-\dfrac {1} {z\`} = z$.
    $2zz\` = -2$.
    $\pow{z}{2} = -2x + C$.
    $\pow{z}{2} + 2x = C$.

    \item $C\pow{x}{2} + \pow{y}{2} = 1$.
    $C = \dfrac {1 - \pow{y}{2}} {\pow{x}{2}}$.
    $0 = \dfrac {-2yy\`\pow{x}{2} - \br{1 - \pow{y}{2}}2x} {\pow{x}{4}}$.
    $-2yy\`\pow{x}{2} - 2x + 2x\pow{y}{2} = 0$.
    $-y\` = \dfrac {1 - \pow{y}{2}} {yx}$.
    $\dfrac {1} {z\`} = \dfrac {1 - \pow{z}{2}} {z x}$.
    $z\` = \dfrac {z x} {1 - \pow{z}{2}}$.
    Среди решений есть $z = 0$.
    $\dfrac {\br{1 - \pow{z}{2}} z\`} {z} = x$.

    $\tint \dfrac {1 - \pow{z}{2}} {z} \di z
    = \tint \br{\dfrac {1} {z} - z} \di z
    = \ln\vbr{z} - \dfrac {1} {2} \pow{z}{2} + C$, так что
    $2\ln\vbr{z} - \pow{z}{2} = \pow{x}{2} + C$.
    Выразить из этой записи $z = z\br{x}$ в элементарных функциях для произвольного $C$ не получится.
\end{enumerate}
