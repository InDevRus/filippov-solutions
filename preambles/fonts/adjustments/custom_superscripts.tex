\newcommand{\subpow}[3]{
    \directlua{
        local base = "\luaescapestring{\unexpanded{#1}}"
        base = base:gsub(" +$", "")
        local subscript = "\luaescapestring{\unexpanded{#2}}"
        local superscript = "\luaescapestring{\unexpanded{#3}}"
        local shared_kerning = math.max(subkerns[base] or 0, superkerns[base] or 0)
        local subkerning = shared_kerning + (subscript_compensations[base] or 0)
        tex.print("{" .. base .. "}_{\\kern" .. subkerning .. "em " .. subscript .. "}^{\\kern".. shared_kerning .. "em " .. superscript .. "}")
    }
}

\newcommand{\pow}[3][-1]{
    \directlua{
        local base = "\luaescapestring{\unexpanded{#2}}"
        base = base:gsub(" +$", "")
        local superscript = "\luaescapestring{\unexpanded{#3}}"
        local kerning = 0
        if #1 >= 0 then
            kerning = #1
        else
            local base_tail = base:sub(-1)
            if base_tail:match("[0-9]") then
                kerning = superkerns[base_tail] or 0
            else
                kerning = superkerns[base] or 0.05
            end
        end
        tex.print("{" .. base .. "}^{\\kern".. kerning .. "em " .. superscript .. "}")
    }
}
