\newcommand{\skewaccent}[2]{
    \directlua{
        local skews = {
            a = 2, b = 2, c = 4, d = 2, e = 5, f = 3, i = 7, j = 7, h = 2, k = 2, l = 7, m = -3, n = 1, o = 2, p = 2, q = 2,
            r = 3, s = 3, t = 7, u = 2, v = 2, w = -1, x = 2, y = 2, z = 3,
            A = 5, B = 1, D = -2, E = 1, F = 1, G = -1, H = -3, I = 4, J = 4, K = -2, M = -5, N = -3, P = 1, R = 1, S = 2,
            U = -3, V = -1, W = -5, X = -2, Y = -1,
            ["\\gamma"] = 1, ["\\delta"] = 1, ["\\epsilon"] = 2, ["\\zeta"] = 3, ["\\eta"] = -1, ["\\iota"] = 4,
            ["\\mu"] = -1, ["\\nu"] = 1, ["\\xi"] = 2, ["\\rho"] = 1, ["\\sigma"] = -1, ["\\tau"] = 2, ["\\phi"] = -1,
            ["\\chi"] = 1, ["\\psi"] = -2, ["\\omega"] = -3, ["\\stigma"] = 3,
            ["\\Alpha"] = -4, ["\\Beta"] = -4, ["\\Gamma"] = -1, ["\\Delta"] = -3, ["\\Epsilon"] = -3, ["\\Zeta"] = -1,
            ["\\Eta"] = -4, ["\\Theta"] = -4, ["\\Iota"] = 3, ["\\Kappa"] = -5, ["\\Lambda"] = -4, ["\\Mu"] = -8,
            ["\\Nu"] = -4, ["\\Xi"] = -3, ["\\Omicron"] = -4, ["\\Pi"] = -4, ["\\Rho"] = -2, ["\\Sigma"] = -1,
            ["\\Tau"] = -2, ["\\Upsilon"] = -4, ["\\Phi"] = -4, ["\\Chi"] = -3, ["\\Psi"] = -4, ["\\Omega"] = -4,
            ["\\f"] = 5, ["\\y"] = 2, ["\\j"] = 7
        }
        local char = "\luaescapestring{\unexpanded{#1}}"
        char = char:gsub(" +$", "")
        local amount = skews[char] or 0
        local accent = "\luaescapestring{\unexpanded{#2}}"
        tex.print("\\skew{" .. amount .. "}" .. accent .. "{" .. char .. "}")
    }
}
