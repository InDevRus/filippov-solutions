% Последний вызов \setmathfont меняет общие метрики шрифта, такие как отступы числителя и знаменателя
% от дробной черты, знак радикала и некоторые другие. Ниже предствален патч для шрифта Asana Math,
% который был выбран из-за наиболее удачной реализации знака радикала.
\directlua{
    function patch_asana(fontdata)
        texio.write_nl("log", "message1" .. fontdata.psname)
        if fontdata.psname == "Asana-Math" then
            local fracM = 0.85
            local dfracUpM = 0.88
            local dfracDownM = 0.975

            if fontdata.MathConstants then
                local mc = fontdata.MathConstants
                mc.FractionNumeratorShiftUp = fracM * mc.FractionNumeratorShiftUp
                mc.FractionDenominatorShiftDown = fracM * mc.FractionDenominatorShiftDown
                mc.FractionNumeratorDisplayStyleShiftUp = dfracUpM * mc.FractionNumeratorDisplayStyleShiftUp
                mc.FractionDenominatorDisplayStyleShiftDown = dfracDownM * mc.FractionDenominatorDisplayStyleShiftDown
            end
            if fontdata.mathparameters then
                local mp = fontdata.mathparameters
                mp.FractionNumeratorShiftUp = fracM * mp.FractionNumeratorShiftUp
                mp.FractionDenominatorShiftDown = fracM * mp.FractionDenominatorShiftDown
                mp.FractionNumeratorDisplayStyleShiftUp = dfracUpM * mp.FractionNumeratorDisplayStyleShiftUp
                mp.FractionDenominatorDisplayStyleShiftDown = dfracDownM * mp.FractionDenominatorDisplayStyleShiftDown
            end
        end
    end
    luatexbase.add_to_callback("luaotfload.patch_font", patch_asana, "patch_asana")
}
\setmathfont[range = {"221A-"221C}]{Asana-Math.otf}
